472:     intentNow.sprint = false;
473:     let yawT = s.yaw;
474:     let pitchT = 0;
475:     if (r.arrive.point) {
476:       const p = r.arrive.point;
477:       const de = p.x - s.e, dn = -p.z - s.n;
478:       // A larger compass bearing is a smaller yaw (yaw grows counter-clockwise).
479:       yawT = bearingToYaw(Math.atan2(de, dn) / DEG + (r.arrive.yawOff ?? 0) / DEG);
480:       pitchT = Math.atan2(p.y - s.eyeY, Math.max(Math.hypot(de, dn), 0.1)) + (r.arrive.pitchOff ?? 0);
481:     } else if (r.arrive.yaw !== null) {
482:       yawT = r.arrive.yaw;
483:     }
484:     const k = Math.min(1, dt / Math.max(r.arrive.t, 1e-6));
485:     intentNow.yawDelta = wrapPi(yawT - s.yaw) * k;
486:     intentNow.pitchDelta = (pitchT - s.pitch) * k;
487:     r.arrive.t -= dt;
488:     if (r.arrive.t <= 1e-6) r.done = true;
489:   }
