"""Pure-Python rest-geometry reach study; no Blender or application mutation."""
import json, math
from pathlib import Path
ROOT=Path('/workspace/chicago/chicago/4d')
OUT=Path('/workspace/shared/human-transport-draft')
add=lambda a,b:[x+y for x,y in zip(a,b)]
sub=lambda a,b:[x-y for x,y in zip(a,b)]
rows=[]
for age in (25,35):
 p=ROOT/f'assets/authored/humans/makehuman_hm08/prepared_adult_{age}.json'
 d=json.loads(p.read_text());bones={b['name']:b for b in d['skeleton']}
 pelvis=[0,1.68,.65];shift=sub(pelvis,bones['pelvis']['head_m'])
 record={'prepared_body':p.name,'engineering_age_parameter':age,'height_m':max(v[1] for m in d['meshes'] for v in m['positions_m']),'pelvis_head_target_m':pelvis,'seat_top_m':1.58,'pelvis_clearance_m':.10,'sides':[]}
 for sign,side in [(1,'l'),(-1,'r')]:
  foot=bones['foot_'+side]['head_m'];vertices=[v for m in d['meshes'] for v,w in zip(m['positions_m'],m['weights']) if w.get('foot_'+side,0)+w.get('ball_'+side,0)>=.9]
  lo=[min(v[k] for v in vertices) for k in range(3)];hi=[max(v[k] for v in vertices) for k in range(3)]
  sole_offset=foot[1]-lo[1]
  ankle=[sign*.18,1.12+sole_offset,1.19]
  wrist=[sign*.20,1.92,1.04]
  result={'side':side,'dominant_foot_vertices':len(vertices),'foot_bounds_relative_ankle_m':{'min':sub(lo,foot),'max':sub(hi,foot)},'ankle_target_m':ankle,'wrist_target_m':wrist,'chains':[]}
  posed_lo=add(sub(lo,foot),ankle);posed_hi=add(sub(hi,foot),ankle)
  result['rigid_foot_bounds_at_target_m']={'min':posed_lo,'max':posed_hi}
  result['footboard_projection_within_bounds']=posed_lo[0]>=-.425 and posed_hi[0]<=.425 and posed_lo[2]>=1.12 and posed_hi[2]<=1.42
  for first,second,target in [('thigh_'+side,'calf_'+side,ankle),('upperarm_'+side,'lowerarm_'+side,wrist)]:
   a,b=bones[first],bones[second];origin=add(a['head_m'],shift);l1=math.dist(a['head_m'],a['tail_m']);l2=math.dist(b['head_m'],b['tail_m']);distance=math.dist(origin,target)
   result['chains'].append({'bones':[first,second],'origin_m':origin,'lengths_m':[l1,l2],'distance_m':distance,'min_reach_m':abs(l1-l2),'max_reach_m':l1+l2,'extension_margin_m':l1+l2-distance,'reachable':abs(l1-l2)<distance<l1+l2,'joint_gap_m':math.dist(a['tail_m'],b['head_m'])})
  record['sides'].append(result)
 rows.append(record)
(OUT/'w01-rest-reach.json').write_text(json.dumps({'status':'analytic reach/projection only; not skinned pose, seat contact, rein fit, or riding validation','bodies':rows},indent=2)+'\n')
for r in rows:
 print(r['prepared_body'])
 for s in r['sides']:
  print(s['side'],'ankle',s['ankle_target_m'],'footboard',s['footboard_projection_within_bounds'])
  for c in s['chains']:print(c['bones'],round(c['distance_m'],6),round(c['max_reach_m'],6),round(c['extension_margin_m'],6),c['reachable'])
