mkdir -p /tmp/scripts && cat > /tmp/scripts/qa_t2376.mjs <<'JS'
// T-2376 QA: the Green Tree's new dormer cap in the published app, both viewports.
import http from 'node:http';
import fs from 'node:fs';
import path from 'node:path';
import {chromium} from 'playwright';

const ROOT = path.resolve(process.argv[2]);
const OUT = path.resolve(process.argv[3]);
const ONLY = process.argv[4] || '';
fs.mkdirSync(OUT, {recursive: true});
const TYPES = {'.html':'text/html','.js':'text/javascript','.mjs':'text/javascript','.css':'text/css',
  '.json':'application/json','.glb':'model/gltf-binary','.png':'image/png','.jpg':'image/jpeg',
  '.jpeg':'image/jpeg','.svg':'image/svg+xml','.webp':'image/webp','.ktx2':'image/ktx2',
  '.woff2':'font/woff2','.bin':'application/octet-stream','.geojson':'application/json','.txt':'text/plain'};
let serverErrors = [];
const server = http.createServer((req,res)=>{
  const url = decodeURIComponent(req.url.split('?')[0]);
  let file = path.join(ROOT, url);
  if (fs.existsSync(file) && fs.statSync(file).isDirectory()) file = path.join(file,'index.html');
  if (!file.startsWith(ROOT) || !fs.existsSync(file)) {
    serverErrors.push(`404 ${url}`);
    res.writeHead(404,{'content-type':'text/plain'}); res.end('not found'); return;
  }
  res.writeHead(200,{'content-type':TYPES[path.extname(file)]||'application/octet-stream'});
  fs.createReadStream(file).pipe(res);
});
const PORT = 8731 + (ONLY === 'desktop' ? 1 : 0);
await new Promise(r=>server.listen(PORT,r));
console.log(`serving ${ROOT} on ${PORT}`);

let bad = 0;
for (const [label, viewport, touch] of [
  ['mobile 390x780', {width:390,height:780}, true],
  ['desktop 1280x800', {width:1280,height:800}, false],
].filter(([l]) => !ONLY || l.includes(ONLY))) {
  console.log(`\n=== ${label} ===`);
  const pageErrors=[], consoleErrors=[], netErrors=[];
  serverErrors = [];
  const browser = await chromium.launch({args:['--enable-unsafe-swiftshader']});
  const ctx = await browser.newContext({viewport, hasTouch:touch, isMobile:false, deviceScaleFactor: touch?2:1});
  const page = await ctx.newPage();
  page.on('pageerror', e => pageErrors.push(String(e)));
  page.on('console', m => { if (m.type()==='error') consoleErrors.push(m.text()); });
  page.on('requestfailed', r => netErrors.push(`${r.url()} ${r.failure()?.errorText}`));
  page.on('response', r => { if (r.status()>=400) netErrors.push(`${r.status()} ${r.url()}`); });
  const tag = label.split(' ')[0];
  try {
    await page.goto(`http://127.0.0.1:${PORT}/walk/?year=1835&anchor=green_tree&keep-geometry=1`, {timeout:120000});
    await page.waitForFunction(()=>window.__chicago4d?.ready===true, null, {timeout:240000, polling:1000});
    console.log('  boot: ready');
    await page.waitForTimeout(6000);
    // the Green Tree in the live scene: its mesh, its world bbox, and what it stands on
    const probe = await page.evaluate(() => {
      const a = window.__chicago4d;
      const out = {found:false};
      const names = [];
      a.scene?.traverse?.(o => {
        if (o.name) names.push(o.name);
        if (o.name && o.name.includes('green_tree_tavern')) {
          out.found = true; out.name = o.name;
          if (o.geometry) {
            o.geometry.computeBoundingBox?.();
            const b = o.geometry.boundingBox;
            if (b) out.localBox = [[b.min.x,b.min.y,b.min.z],[b.max.x,b.max.y,b.max.z]];
            out.tris = (o.geometry.index ? o.geometry.index.count : o.geometry.attributes?.position?.count ?? 0)/3;
          }
        }
      });
      out.nodeCount = names.length;
      out.anchor = a.spawn ?? null;
      return out;
    });
    console.log('  green tree in scene:', JSON.stringify(probe).slice(0,400));
    await page.screenshot({path: path.join(OUT, `${tag}-normal.png`)});
    console.log(`  shot: ${tag}-normal.png`);
    // By grade — the Confidence view's own preset, bound to C
    await page.keyboard.press('c');
    await page.waitForTimeout(5000);
    await page.screenshot({path: path.join(OUT, `${tag}-bygrade.png`)});
    console.log(`  shot: ${tag}-bygrade.png`);
    await page.keyboard.press('c');
    await page.waitForTimeout(3000);
    await page.screenshot({path: path.join(OUT, `${tag}-normal-again.png`)});
  } catch (e) {
    console.log('  THREW:', String(e).slice(0,300)); bad += 1;
  }
  console.log(`  pageerrors ${pageErrors.length}, console errors ${consoleErrors.length}, network ${netErrors.length}, server 404s ${serverErrors.length}`);
  for (const e of pageErrors.slice(0,5)) console.log('   PAGEERROR', e.slice(0,200));
  for (const e of consoleErrors.slice(0,5)) console.log('   CONSOLE', e.slice(0,200));
  for (const e of netErrors.slice(0,5)) console.log('   NET', e.slice(0,200));
  for (const e of serverErrors.slice(0,5)) console.log('   SERVER', e.slice(0,200));
  if (pageErrors.length||consoleErrors.length||netErrors.length||serverErrors.length) bad += 1;
  await browser.close();
}
server.close();
console.log(bad ? `\nQA FAIL (${bad})` : '\nQA PASS — zero page, console, network and server errors');
process.exit(bad?1:0);
JS
